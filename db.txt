CREATE DATABASE QuanLyBenhNhanNoiTru1234;

USE QuanLyBenhNhanNoiTru1234;

CREATE TABLE TAIKHOAN (
    TenDangNhap NVARCHAR(50)  NOT NULL PRIMARY KEY,
    MatKhau     NVARCHAR(100) NOT NULL,
    HoTen       NVARCHAR(100) NOT NULL,
    VaiTro      NVARCHAR(30)  NOT NULL DEFAULT N'Admin',
    TrangThai   BIT           NOT NULL DEFAULT 1
);

CREATE TABLE KHOA (
    MaKhoa  NVARCHAR(10)  NOT NULL PRIMARY KEY,
    TenKhoa NVARCHAR(100) NOT NULL
);

CREATE TABLE PHONG (
    MaPhong   NVARCHAR(10)  NOT NULL PRIMARY KEY,
    TenPhong  NVARCHAR(100) NOT NULL,
    MaKhoa    NVARCHAR(10)  NOT NULL REFERENCES KHOA(MaKhoa),
    DonGia    DECIMAL(18,0) NOT NULL DEFAULT 0 CHECK (DonGia >= 0),   -- tiền phòng / ngày
    TrangThai BIT           NOT NULL DEFAULT 1                        -- 1 = đang hoạt động
);

CREATE TABLE GIUONG (
    MaGiuong  NVARCHAR(15) NOT NULL PRIMARY KEY,
    MaPhong   NVARCHAR(10) NOT NULL REFERENCES PHONG(MaPhong),
    SoGiuong  INT          NOT NULL,
    TrangThai NVARCHAR(20) NOT NULL DEFAULT N'Trống'
              CHECK (TrangThai IN (N'Trống', N'Đang sử dụng'))
);

CREATE TABLE BACSI (
    MaBS         NVARCHAR(10)  NOT NULL PRIMARY KEY,
    HoTen        NVARCHAR(100) NOT NULL,
    NgaySinh     DATE          NULL,
    GioiTinh     NVARCHAR(5)   NULL,
    ChuyenKhoa   NVARCHAR(100) NULL,
    SoDienThoai  NVARCHAR(20)  NULL,
    Email        NVARCHAR(100) NULL,
    MaKhoa       NVARCHAR(10)  NULL REFERENCES KHOA(MaKhoa),
    TrangThai    BIT           NOT NULL DEFAULT 1                     -- 1 = đang làm việc
);

CREATE TABLE BENHNHAN (
    MaBN          NVARCHAR(10)  NOT NULL PRIMARY KEY,
    HoTen         NVARCHAR(100) NOT NULL,
    NgaySinh      DATE          NULL,
    GioiTinh      NVARCHAR(5)   NULL,
    CCCD          NVARCHAR(12)  NULL,
    SoDienThoai   NVARCHAR(20)  NULL,
    DiaChi        NVARCHAR(200) NULL,
    NgheNghiep    NVARCHAR(100) NULL,
    NguoiThan     NVARCHAR(100) NULL,
    SDTNguoiThan  NVARCHAR(20)  NULL,
    TinhTrang     NVARCHAR(500) NULL                                  -- tình trạng bệnh hiện tại
);

CREATE TABLE NHAPVIEN (
    MaNhapVien    NVARCHAR(10)  NOT NULL PRIMARY KEY,
    MaBN          NVARCHAR(10)  NOT NULL REFERENCES BENHNHAN(MaBN),
    MaBS          NVARCHAR(10)  NULL REFERENCES BACSI(MaBS),
    MaGiuong      NVARCHAR(15)  NULL REFERENCES GIUONG(MaGiuong),
    NgayNhapVien  DATETIME      NOT NULL DEFAULT GETDATE(),
    NgayRaVien    DATETIME      NULL,
    LyDoNhapVien  NVARCHAR(500) NULL,
    TrangThai     NVARCHAR(20)  NOT NULL DEFAULT N'Đang điều trị'
                  CHECK (TrangThai IN (N'Đang điều trị', N'Đã ra viện', N'Hủy')),
    GhiChu        NVARCHAR(500) NULL
);

CREATE UNIQUE INDEX UX_BENHNHAN_CCCD ON BENHNHAN(CCCD) WHERE CCCD IS NOT NULL;

CREATE UNIQUE INDEX UX_NHAPVIEN_BN_DANGDT ON NHAPVIEN(MaBN)
    WHERE TrangThai = N'Đang điều trị';
CREATE UNIQUE INDEX UX_NHAPVIEN_GIUONG_DANGDT ON NHAPVIEN(MaGiuong)
    WHERE TrangThai = N'Đang điều trị' AND MaGiuong IS NOT NULL;
    
CREATE TABLE BENHAN (
    MaBA          NVARCHAR(10)  NOT NULL PRIMARY KEY,
    MaNhapVien    NVARCHAR(10)  NOT NULL REFERENCES NHAPVIEN(MaNhapVien),
    NgayLap       DATETIME      NOT NULL DEFAULT GETDATE(),
    TrieuChung    NVARCHAR(1000) NULL,
    ChanDoan      NVARCHAR(1000) NULL,
    KetQuaKham    NVARCHAR(1000) NULL,
    HuongDieuTri  NVARCHAR(1000) NULL
);
 
CREATE TABLE THUOC (
    MaThuoc    NVARCHAR(10)  NOT NULL PRIMARY KEY,
    TenThuoc   NVARCHAR(100) NOT NULL,
    DonVi      NVARCHAR(20)  NULL,
    DonGia     DECIMAL(18,0) NOT NULL DEFAULT 0 CHECK (DonGia >= 0),
    SoLuongTon INT           NOT NULL DEFAULT 0 CHECK (SoLuongTon >= 0),
    TrangThai  BIT           NOT NULL DEFAULT 1
);
 
CREATE TABLE DONTHUOC (
    MaDon      NVARCHAR(10)  NOT NULL PRIMARY KEY,
    MaBA       NVARCHAR(10)  NOT NULL REFERENCES BENHAN(MaBA),
    NgayKeDon  DATETIME      NOT NULL DEFAULT GETDATE(),
    GhiChu     NVARCHAR(500) NULL
);
 
CREATE TABLE CT_DONTHUOC (
    MaDon     NVARCHAR(10)  NOT NULL REFERENCES DONTHUOC(MaDon),
    MaThuoc   NVARCHAR(10)  NOT NULL REFERENCES THUOC(MaThuoc),
    SoLuong   INT           NOT NULL CHECK (SoLuong > 0),
    LieuDung  NVARCHAR(200) NULL,
    CachDung  NVARCHAR(200) NULL,
    DonGia    DECIMAL(18,0) NOT NULL,                                  -- giá tại thời điểm kê đơn
    ThanhTien AS (SoLuong * DonGia) PERSISTED,
    PRIMARY KEY (MaDon, MaThuoc)
);
 
CREATE TABLE VIENPHI (
    MaVP        NVARCHAR(10)  NOT NULL PRIMARY KEY,
    MaNhapVien  NVARCHAR(10)  NOT NULL UNIQUE REFERENCES NHAPVIEN(MaNhapVien),
    TienPhong   DECIMAL(18,0) NOT NULL DEFAULT 0,
    TienThuoc   DECIMAL(18,0) NOT NULL DEFAULT 0,
    TienKham    DECIMAL(18,0) NOT NULL DEFAULT 0,
    TienDichVu  DECIMAL(18,0) NOT NULL DEFAULT 0,
    TongTien    AS (TienPhong + TienThuoc + TienKham + TienDichVu) PERSISTED,
    NgayLap     DATETIME      NOT NULL DEFAULT GETDATE(),
    TrangThai   NVARCHAR(20)  NOT NULL DEFAULT N'Chưa thanh toán'
                CHECK (TrangThai IN (N'Chưa thanh toán', N'Đã thanh toán'))
);

-- Trừ tồn kho theo các đơn đã kê (giống cách chương trình tự trừ)
UPDATE t
SET t.SoLuongTon = t.SoLuongTon - s.Tong
FROM THUOC t
JOIN (SELECT MaThuoc, SUM(SoLuong) AS Tong FROM CT_DONTHUOC GROUP BY MaThuoc) s ON s.MaThuoc = t.MaThuoc;
